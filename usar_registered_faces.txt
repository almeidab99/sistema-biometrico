        // Conectarse al servidor ESP32-CAM a traves de WebSocket

        const ws = new WebSocket(WS_URL); 
        // Funcion para mostrar caras en la lista
        function addFaceToList(faceName) {
            const faceList = document.getElementById('faceList');
            const listItem = document.createElement('li');
            listItem.textContent = faceName;
            faceList.appendChild(listItem);
        }

        // Manejar mensajes del servidor
        ws.onmessage = message => {     
            if (typeof message.data === "string") {    
                if (message.data.substr(0, 8) == "listface") {   
             // Si el mensaje comienza con 'listface', agrega la cara a la lista
                    const faceName = message.data.substr(8);
                    addFaceToList(faceName);
        }
    }
};

        // Manejar errores de WebSocket
        ws.onerror = (error) => {
            console.error('Error de WebSocket:', error);
        };

        // Manejar cierre de WebSocket
        ws.onclose = () => {
            console.log('Conexion WebSocket cerrada');
        };
    </script>
